nxn-engine v1.1.0

Az nxn-engine kezeli a jármű motorának indítását/leállítását, HP-alapú sérülését, sérülés-romlás (degradáció) rendszerét és szinkronizál az nxn-vehicle-hud-dal. Export API-ja előkészítve az nxn-keys (kulcsrendszer) és nxn-hotwire (autólopás) resource-okhoz.

Integrációk: nxn-vehicle-hud (setEngineState), nxn-notify, nxn-seatbelt-extras (collision event)

Degradáció (sérülés-romlás) rendszer

Minél sérültebb a motor, annál rosszabbul működik: csökkent teljesítmény, ritmikus leállogatás (stutter), füst és szikra effektek. Minden alrendszer a Config.Degradation blokkban konfigurálható.

Stutter logika: HP < stutter.threshold (alap: 50%) esetén minden checkInterval másodpercben eséllyel leáll a motor. Az esély arányos: minél alacsonyabb a HP, annál sűrűbb a kihagy. Leállás után restartChance valószínűséggel újraindul.

-- nxn-keys/client.lua inicializalas:

-- 1. Auth callback regisztralasa - ez fut le minden inditasnal
exports['nxn-engine']:registerStartAuthCallback(function(vehicleEntity)
    local plate = GetVehicleNumberPlateText(vehicleEntity)
    -- Ellenorizd hogy a jatekosnak van-e kulcsa ehhez a rendszamhoz
    local hasKey = NXN.Keys.HasKeyForPlate(plate)
    NXN.Keys.Log(('Auth check: plate=%s hasKey=%s'):format(plate, tostring(hasKey)))
    return hasKey
end)

-- 2. Kulcs nelkul le is zaroljuk (pl. jarmu ellopas utan):
exports['nxn-engine']:setLocked(true)

-- 3. Kulcs visszaadasa utan feloldas:
exports['nxn-engine']:setLocked(false)
-- nxn-hotwire/client.lua:

-- Hotwire siker utan motor inditasa:
local ok = exports['nxn-engine']:hotwireStart()
if ok then
    exports['nxn-notify']:success('Sikeres hotwire!')
else
    exports['nxn-notify']:danger('A motor tul sulyosan serult.')
end

-- Motor allapot ellenorzese hotwire elott:
local hp = exports['nxn-engine']:getEngineHP()
if hp < 10 then
    exports['nxn-notify']:danger('A motor tul torott a hotwire-hoz.')
end
-- Motor serules esemeny figyelese:
AddEventHandler('nxn-engine:damaged', function(data)
    print(('Motor serules: reason=%s dmg=%.1f hp=%.1f'):format(
        data.reason, data.dmg, data.hp
    ))
    if data.hp <= 0 then
        TriggerServerEvent('nxn-mechanic:server:engineDestroyed', {
            vehicle = data.vehicle
        })
    end
end)

-- Motor HP manualis allitasa (pl. szerelo utan):
exports['nxn-engine']:setEngineHP(100)  -- teli repair
exports['nxn-engine']:setEngineHP(50)   -- felig megjavitva
-- Degradacios allapot folyamatos figyelese (pl. HUD frissiteshez):

CreateThread(function()
    while true do
        Wait(1000)
        if exports['nxn-engine']:isInVehicle() then
            local hp   = exports['nxn-engine']:getEngineHP()
            local perf = exports['nxn-engine']:getPerfModifier()
            local stut = exports['nxn-engine']:isStuttering()

            -- Kiiras konzolra (debug)
            print(('HP: %.1f%% | Teljesitmeny: %.0f%% | Stutter: %s'):format(
                hp, perf * 100, tostring(stut)
            ))

            -- Pl. masik script ertesitese a teljesitmenyrol:
            -- exports['nxn-vehicle-hud']:setPerfBar(perf)
        end
    end
end)

-- Stutter esemenyek figyelese (pl. hangeffekt lejatszas):
AddEventHandler('nxn-engine:stutter:stall', function(data)
    -- Motor kihagyott data.duration masodpercre
    -- Pl. kikapcsolo hang lejatszasa:
    -- PlaySoundFrontend(-1, 'ENGINE_OFF', 'HUD_FRONTEND_DEFAULT_SOUNDSET', true)
    print(('Stutter: motor leall %.1f mp-re, HP=%.1f%%'):format(data.duration, data.hp))
end)

AddEventHandler('nxn-engine:stutter:failed', function(data)
    -- Motor nem indult ujra stutter utan
    exports['nxn-notify']:danger('A motor teljesen kikeszult – szerviz szukseges!')
end)
-- nxn-mechanic / szerviz script peldaja:

-- Teljes javitas utan motor HP visszaallitasa ES degradacio reset:
local function FullRepair()
    exports['nxn-engine']:setEngineHP(100)
    exports['nxn-engine']:resetDegradation()
    exports['nxn-notify']:success('Motor teljesen megjavitva!')
end

-- Reszleges javitas (pl. 70%-ig):
local function PartialRepair(targetHP)
    exports['nxn-engine']:setEngineHP(targetHP)
    -- Ha a cel HP mar a stutter kuszob felett van,
    -- a degradacio automatikusan csokkent kovetkezo ticknel,
    -- de a resetDegradation() azonnal visszaallitja:
    if targetHP >= 50 then
        exports['nxn-engine']:resetDegradation()
    end
    exports['nxn-notify']:success(('Motor %d%%-ra megjavitva.'):format(targetHP))
end

-- Peldahivas:
FullRepair()
PartialRepair(70)

Debug

Config.Debug = true esetén minden motor-esemény, RPM mintavétel, HUD szinkron, sérülés, degradáció tick és stutter check megjelenik [nxn-engine] [DEBUG] prefixszel a konzolban. Figyelj a PerfDeg:, Stutter check: és Effekt sorokra a degradáció diagnosztikához.
