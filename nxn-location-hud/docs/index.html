nxn-location-hud v1.0.0

Az nxn-location-hud a Nexon szerver moduláris lokációs HUD-ja. Körzet, utca, koordináta, zóna, veszely szint, körözés és játékos státusz modulokból épül fel. Az nxn-gang, nxn-dispatch és nxn-wantedstatus exportokon keresztül tölti fel az adatokat.

Függőség: nincs kötelező függőség, önállóan fut.

Modul kapcsolás
Bővítés
-- nxn-gang / client.lua:
AddEventHandler('nxn-gang:zoneEntered', function(data)
    exports['nxn-location-hud']:setModule('zone', true)
    exports['nxn-location-hud']:setZone(
        data.zoneName,   -- 'Davis'
        data.gangName,   -- 'Ballas'
        data.color       -- '#8b00ff'
    )
end)

AddEventHandler('nxn-gang:zoneLeft', function()
    exports['nxn-location-hud']:clearZone()
end)
-- nxn-dispatch / client.lua:
AddEventHandler('nxn-dispatch:alert', function(level, label)
    -- level: 1-5, label: 'AKTIV LOVOLDOZES'
    exports['nxn-location-hud']:setModule('danger', true)
    exports['nxn-location-hud']:setDanger(level, label)
end)

AddEventHandler('nxn-dispatch:clear', function()
    exports['nxn-location-hud']:setDanger(0)
end)
-- nxn-wantedstatus / client.lua:
RegisterNetEvent('nxn-wantedstatus:client:update')
AddEventHandler('nxn-wantedstatus:client:update', function(wanted, stars, label)
    exports['nxn-location-hud']:setModule('wanted', true)
    exports['nxn-location-hud']:setWanted(wanted, stars, label)
end)
-- Játékos státusz modul bekapcsolása
exports['nxn-location-hud']:setModule('playerstatus', true)

-- HUD elrejtése cutscene alatt
exports['nxn-location-hud']:setVisible(false)

-- Pozíció váltás
exports['nxn-location-hud']:setPosition('bottom-right')
