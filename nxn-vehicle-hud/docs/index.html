nxn-vehicle-hud v1.0.0

Az nxn-vehicle-hud a Nexon szerver moduláris jármű HUD-ja. Automatikusan megjelenik járműbe szállásnál és eltűnik kiszállásnál. Minden modul külön fájlban él, könnyen bővíthető. Az opcionális modulokat (fuel, seatbelt, siren, engine) a dedikált külső resourceok töltik fel.

Önálló resource – nincs kötelező függősége, az nxn-notify-jal és nxn-hud-dal párhuzamosan fut.

Bővítés: új modul
-- nxn-fuel / client.lua
AddEventHandler('nxn-fuel:updated', function(level)
    -- fuel modul bekapcsolasa ha meg nem lenne
    exports['nxn-vehicle-hud']:setModule('fuel', true)
    exports['nxn-vehicle-hud']:setFuel(level)
end)
-- nxn-seatbelt / client.lua
AddEventHandler('nxn-seatbelt:changed', function(fastened)
    exports['nxn-vehicle-hud']:setModule('seatbelt', true)
    exports['nxn-vehicle-hud']:setSeatbelt(fastened)
end)
-- nxn-sirencontrol / client.lua
AddEventHandler('nxn-sirencontrol:changed', function(active, mode)
    exports['nxn-vehicle-hud']:setModule('siren', true)
    exports['nxn-vehicle-hud']:setSiren(active, mode)
end)
-- pl. mode = 'Code 2', 'Code 3', 'Wail' stb.
-- nxn-engine / client.lua (motor leállítás / beindítás esemény)
AddEventHandler('nxn-engine:stateChanged', function(state)
    -- state: 'ok' | 'damaged' | 'critical' | 'off'
    exports['nxn-vehicle-hud']:setEngineState(state)
end)
-- HUD elrejtése (pl. cutscene)
exports['nxn-vehicle-hud']:setVisible(false)

-- Üzemanyag modul kikapcsolása
exports['nxn-vehicle-hud']:setModule('fuel', false)

-- Pozíció váltás
exports['nxn-vehicle-hud']:setPosition('bottom-left')
