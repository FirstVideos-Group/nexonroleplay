nxn-bank v1.0.0

Az nxn-bank az nxn-finance pénzügyi motorra épülő dedikált bankrendszer. ATM-eken és bankfiókokban kezeli a befizetést, felvétet, játékosok közötti átutalást, és lapozható tranzakciónaplót biztosít.

├── shared.lua
├── client.lua
├── server.lua
└── html/
    ├── index.html
    ├── style.css
    └── app.js

Függőségek

Kötelező nxn-finance – minden pénzmozgás ezen keresztül történik
Kötelező nxn-database – tranzakciónapló tábla, identifier
Kötelező nxn-notify – visszajelzések
Kötelező nxn-identity – játékos neve átutalásnál
Opcionális nxn-hud – egyenleg badge frissítés

Export használati példák

Szerver – befizetés
Szerver – átutalás
Szerver – napló
Kliens – panel
-- Játékos fizet be $1000-t (pl. munkabér érkezésekor)
local ok = exports['nxn-bank']:deposit(src, 1000)
if ok then
    print('Befizetés sikeres')
end
-- Átutalás másik játékosnak
local ok = exports['nxn-bank']:transfer(src, targetSrc, 500, 'Autó ára')
if not ok then
    -- nincs elég pénz vagy hibás célszemély
end
-- Tranzakciónapló lekérése (szerver oldali script)
local identifier = exports['nxn-database']:getIdentifier(src)
local txList = exports['nxn-bank']:getTransactions(identifier, 1)
for _, tx in ipairs(txList) do
    print(tx.type, tx.amount, tx.description)
end
-- ATM panel megnyitása kliensről (pl. nxn-target interakció)
exports['nxn-bank']:openATM()

-- Bankpult (teljes panel: átutalás + napló)
exports['nxn-bank']:openBankCounter()

-- Bezárás
exports['nxn-bank']:closeBank()

Adatbázis tábla

CREATE TABLE IF NOT EXISTS `nxn_bank_transactions` (
    `id`          INT UNSIGNED NOT NULL AUTO_INCREMENT,
    `identifier`  VARCHAR(64)  NOT NULL,
    `type`        ENUM('deposit','withdraw','transfer_in','transfer_out','fine') NOT NULL,
    `amount`      INT          NOT NULL,
    `description` VARCHAR(255) DEFAULT NULL,
    `target_id`   VARCHAR(64)  DEFAULT NULL,
    `created_at`  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    INDEX `idx_identifier` (`identifier`),
    INDEX `idx_created_at` (`created_at`)
);
