nxn-trunk v1.0.0

Az nxn-trunk egy jármű csomagtartó-kezelő script, amely az nxn-inventory rendszerére épít. Két-paneles NUI felületen teszi lehetővé az inventory â csomagtartó átmozgatást, járműtípusonként állítható kapacitással.

└── html/
    ├── index.html
    ├── style.css
    └── app.js

Függőségek

Kötelező nxn-inventory – addItem, removeItem, hasItem, getInventory, syncInventory
Kötelező nxn-database – registerTable, nxn_trunks tábla
Kötelező nxn-notify – visszajelzések
Opcionális nxn-keys / nxn-engine – zár-ellenőrzés (jövőbeli integráció)

Export használati példák

Szerver – trunk adat
Szerver – item mozgás
Kliens – panel
Szerver – belső hook
-- Trunk tartalmának lekérése
local trunk = exports['nxn-trunk']:getTrunk('ABC123')
if trunk then
    for _, item in ipairs(trunk.items) do
        print(item.name, item.count, item.weight)
    end
end

-- Jelenlegi súly
local w = exports['nxn-trunk']:getTrunkWeight('ABC123')
print('Trunk súly: ' .. w .. ' kg')
-- Item hozzáadása csomagtartóhoz (pl. loot spawnalnál)
local ok, msg = exports['nxn-trunk']:addToTrunk('ABC123', 'water_bottle', 3, 0.5)
if not ok then print('Hiba: ' .. msg) end

-- Item eltávolítása
exports['nxn-trunk']:removeFromTrunk('ABC123', 'water_bottle', 1)

-- Van-e benne?
if exports['nxn-trunk']:hasTrunkItem('ABC123', 'bandage', 2) then
    print('Van legalább 2 kötöszer')
end
-- Csomagtartó megnyitása entity handle-lel
local vehicle = GetVehiclePedIsIn(PlayerPedId(), false)
exports['nxn-trunk']:openTrunk(vehicle)

-- Bezárás
exports['nxn-trunk']:closeTrunk()

-- Legközelebbi jármű hátsó részénél
local v = exports['nxn-trunk']:getNearbyVehicle()
if v then print('Közeli jármű: ' .. v) end
-- Figyeljél az item mozgás esemire más scriptekből:
AddEventHandler('nxn-trunk:server:itemMoved', function(src, plate, itemName, count, direction)
    print(('Játékos %d átrakhott: %d x %s (%s) plate=%s'):format(
        src, count, itemName, direction, plate
    ))
end)

-- Trunk megnyitás event:
AddEventHandler('nxn-trunk:server:opened', function(src, plate)
    print(('Játékos %d megnyitotta: %s'):format(src, plate))
end)

Adatbázis tábla

CREATE TABLE IF NOT EXISTS `nxn_trunks` (
    `id`         INT UNSIGNED NOT NULL AUTO_INCREMENT,
    `plate`      VARCHAR(16)  NOT NULL UNIQUE,
    `items`      JSON         NOT NULL DEFAULT '[]',
    `max_weight` FLOAT        NOT NULL DEFAULT 50.0,
    `updated_at` DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    INDEX `idx_plate` (`plate`)
);
