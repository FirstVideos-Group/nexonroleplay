nxn-delivery v1.0.0

Az nxn-delivery a Nexon RP szerver első önálló, játékható munkája – egy teljesítményalapú szállítási/futármunka rendszer. A játékosok csomagokat szállítanak diszpécserponttól célállomásra, jutalmat kapnak az elvégzett feladatokért.

Működési folyamat

1. Játékos diszpécser NPC-hez megy (delivery munkakör + aktív műszak szükséges)
2. "Feladat kérése" → szerver generál véletlenszerű célpontot és kategóriát
3. PICKUP marker jelenik meg a diszpécser közelében → [E] csomag felvétele (animációval)
4. DROPOFF blip + marker jelenik meg a célállomáson → [E] leadás
5. Szerver ellenőrzi az időt → jutalom kiszámítása (alap + távolság + esetleges bónusz)
6. Fizetés az nxn-finance-on keresztül, DB mentés, fáradtság növelés

-- Van-e aktív feladat?
if exports['nxn-delivery']:hasActiveTask(src) then
    print('Aktív szállítás van.')
end

-- Feladat adatai
local task = exports['nxn-delivery']:getActiveTask(src)
if task then
    print('Kategória: ' .. task.category)
    print('Jutalom: '   .. task.reward .. ' Ft')
end

-- Teljesített feladatok száma a műszakban
local count = exports['nxn-delivery']:getTaskCount(src)
print('Feladatok: ' .. count)

-- Összkereset a műszakban
local earned = exports['nxn-delivery']:getTotalEarned(src)
print('Keresett: ' .. earned .. ' Ft')

-- Feladat lemondása
exports['nxn-delivery']:cancelTask(src, 'admin')

-- Napi toplista (top 5)
local lb = exports['nxn-delivery']:getLeaderboard(5)
for _, row in ipairs(lb) do
    print(row.identifier, row.total_tasks, row.total_earned)
end
-- Van-e aktív feladat lokálisan?
if exports['nxn-delivery']:hasActiveTask() then
    print('Aktív szállítás fut.')
end

-- Feladat adatok kliensen
local task = exports['nxn-delivery']:getLocalTask()
if task then
    print('Célállomás: ' .. task.target.label)
end
-- Szállítás teljesítve (más scriptből figyelve)
AddEventHandler('nxn-delivery:server:taskCompleted', function(src, taskId, reward, bonus, timeTaken)
    print(('Játékos #%d teljesítette a feladatot. Jutalom: %d Ft, bónusz: %s'):format(
        src, reward, tostring(bonus)))
end)

-- Szállítás indult
AddEventHandler('nxn-delivery:server:taskStarted', function(src, taskId, category, target, reward)
    print(('Játékos #%d szállítást indított: %s → %s'):format(src, category, target.label))
end)

-- Szállítás sikertelen
AddEventHandler('nxn-delivery:server:taskFailed', function(src, taskId, reason)
    print(('Játékos #%d szállítása sikertelen: %s'):format(src, reason))
end)

Adatbázis

CREATE TABLE IF NOT EXISTS `nxn_delivery_tasks` (
    `id`            INT UNSIGNED   NOT NULL AUTO_INCREMENT,
    `identifier`    VARCHAR(100)   NOT NULL,
    `category`      VARCHAR(20)    NOT NULL,
    `dispatch_id`   VARCHAR(50)    NOT NULL,
    `target_label`  VARCHAR(100)   NOT NULL,
    `target_coords` VARCHAR(100)   NOT NULL,
    `reward`        INT UNSIGNED   NOT NULL DEFAULT 0,
    `distance_m`    FLOAT          DEFAULT NULL,
    `time_limit`    INT UNSIGNED   NOT NULL,
    `time_taken`    INT UNSIGNED   DEFAULT NULL,
    `status`        ENUM('active','completed','failed','cancelled') NOT NULL DEFAULT 'active',
    `bonus_applied` TINYINT(1)     NOT NULL DEFAULT 0,
    `started_at`    DATETIME       NOT NULL,
    `completed_at`  DATETIME       DEFAULT NULL,
    PRIMARY KEY (`id`),
    INDEX `idx_delivery_ident`  (`identifier`),
    INDEX `idx_delivery_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
