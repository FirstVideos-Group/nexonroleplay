nxn-vehicles v1.0.0

⚠️ Fontos: Az nxn-vehicles kizárólag az adat- és eseményréteget kezeli. A garázs UI, spawn/despawn logika az nxn-garage scriptben valósíul meg, amely az itt definiált exportokat hívja.

Az nxn-vehicles nyílvantartja a játékosok tulajdonában lévő járműveket, figyeli a járműbe lépésést, ellenőrzi a jogosítványokat és perzisztálja a motor HP-t.

Használati példák

nxn-garage – spawn
nxn-licenses – belépés hook
Jármű lekérés / tulajdonos
Motor HP perzisztencia
-- nxn-garage/server.lua – Jármű spawnolása előtt:
local plate = exports['nxn-vehicles']:addVehicle(identifier, 'adder', 'Adder', 14)
if plate then
    exports['nxn-vehicles']:setStored(plate, false)
    local hp = exports['nxn-vehicles']:getEngineHP(plate)
    TriggerClientEvent('nxn-vehicles:client:engineHPSync', src, { plate = plate, hp = hp })
    -- nxn-keys kulcs adása:
    if GetResourceState('nxn-keys') == 'started' then
        exports['nxn-keys']:giveKey(src, plate)
    end
    TriggerEvent('nxn-vehicles:server:spawned', src, plate, 'adder')
end
-- nxn-licenses/server.lua – Járműbe lépés jogosítvány-ellenőrzés:
AddEventHandler('nxn-vehicles:server:entered', function(src, plate, vehicleClass)
    local licType = exports['nxn-vehicles']:getLicenseTypeForClass(vehicleClass)
    if licType then
        local ok = exports['nxn-licenses']:hasLicense(src, licType)
        if not ok then
            -- Figyelmeztetés a játékosnak
            TriggerClientEvent('nxn-notify:client:warning', src,
                'Nincs érvényes jogosítványod!')
        end
    end
end)
-- Játékos járműveinek listázása:
local ident    = exports['nxn-database']:getIdentifier(src)
local vehicles = exports['nxn-vehicles']:getVehicles(ident)
for _, v in ipairs(vehicles) do
    print(v.plate, v.model, v.stored)
end

-- Tulajdonos-ellenőrzés:
if exports['nxn-vehicles']:isOwner(src, 'ABC-12-CD') then
    print('Tulajdonos!')
end
-- Despawn előtt HP mentés:
exports['nxn-vehicles']:saveEngineHP(plate, currentHP)
exports['nxn-vehicles']:setStored(plate, true)
TriggerEvent('nxn-vehicles:server:despawned', src, plate)

-- Spawn után HP visszatöltés:
local hp = exports['nxn-vehicles']:getEngineHP(plate)
TriggerClientEvent('nxn-vehicles:client:engineHPSync', src, { plate = plate, hp = hp })

Adatbázis táblák

-- nxn_vehicles: jármű tulajdon
CREATE TABLE `nxn_vehicles` (
    id, identifier, plate UNIQUE, model, label, class,
    mods JSON, garage, impounded, stored, created_at
);

-- nxn_vehicle_engine_hp: motor HP perzisztencia (#125)
CREATE TABLE `nxn_vehicle_engine_hp` (
    plate PRIMARY KEY, hp_percent FLOAT, updated_at
);
